    /// Minimal in-code string table. Column order: en, zh-Hans, ko, es, ja.
    /// Designed to be replaced by the Unity Localization package later (keys stay the same).
    /// </summary>
    public static class LocalizationTable
    {
        public static readonly string[] LanguageCodes = { "en", "zh-Hans", "ko", "es", "ja" };
        public static readonly string[] NativeNames = { "English", "中文", "한국어", "Español", "日本語" };
        public const string DefaultLanguage = "en";

        private static readonly Dictionary<string, string[]> Table = Build();

        public static int IndexOf(string code)
        {
            for (int i = 0; i < LanguageCodes.Length; i++)
                if (LanguageCodes[i] == code) return i;
            return -1;
        }

        public static bool IsSupported(string code) => IndexOf(code) >= 0;

        public static bool TryGet(string languageCode, string key, out string value)
        {
            value = null;
            int idx = IndexOf(languageCode);
            if (idx < 0) idx = 0;
            if (string.IsNullOrEmpty(key) || !Table.TryGetValue(key, out var row))
                return false;
            value = string.IsNullOrEmpty(row[idx]) ? row[0] : row[idx];
            return true;
        }

        /// <summary>Maps Application.systemLanguage to a supported code (spec 5.2).</summary>
        public static string FromSystemLanguage(UnityEngine.SystemLanguage lang)
        {
            switch (lang)
            {
                case UnityEngine.SystemLanguage.Chinese:
                case UnityEngine.SystemLanguage.ChineseSimplified:
                    return "zh-Hans";
                case UnityEngine.SystemLanguage.Korean: return "ko";
                case UnityEngine.SystemLanguage.Spanish: return "es";
                case UnityEngine.SystemLanguage.Japanese: return "ja";
                default: return DefaultLanguage;
            }
        }

        private static void Add(Dictionary<string, string[]> t, string key, string en, string zh, string ko, string es, string ja)
        {
            t[key] = new[] { en, zh, ko, es, ja };
        }
