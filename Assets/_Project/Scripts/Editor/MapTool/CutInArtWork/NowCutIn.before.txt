using UnityEngine;
using UnityEngine.UI;

namespace MyGame
{
    /// <summary>
    /// "Now!" cut-in played when a boss becomes groggy (stunned): time stops, a slanted band sweeps across the screen
    /// showing a live close-up of the player's eyes (determined face), a star glint flashes on the eye, the localized
    /// "Now!" pops up, and time resumes. Built in code on its own overlay canvas like BossHud; runs on unscaled time.
    /// Only one plays at a time. While it plays the game must not change Time.timeScale (InGameController skips pause).
    /// </summary>
    [DefaultExecutionOrder(10000)] // after the animator / auto-blink so the face override wins this frame
    public sealed class NowCutIn : MonoBehaviour
    {
        public const string TextKey = "cutin.now";

        private const float Total = 1.15f;     // seconds of stopped time
        private const float BandIn = 0.14f;
        private const float GlintAt = 0.30f;
        private const float TextAt = 0.20f;
        private const float BandOut = 0.88f;
        private const float BandHeight = 300f;
        private const float BandAngle = -7f;
        private const int RtWidth = 1600, RtHeight = 300;

        private static NowCutIn current;
        public static bool IsPlaying => current != null;

        private float t;
        private float savedTimeScale = 1f;
        private Camera eyeCam;
        private RenderTexture rt;
        private Transform eyeL, eyeR, modelRoot;
        private SkinnedMeshRenderer face;
        private int[] faceShapes;
        private float[] faceWeights;
        private int blinkShape = -1;

        private RectTransform band, eyeImage, text, glintA, glintB, streak;
        private Image dim, flash;
        private RawImage lines;
        private Graphic[] bandGraphics;

        /// <summary>Starts the cut-in for the given player (any transform under which the humanoid model lives).</summary>
        public static void Play(Transform player)
        {
            if (current != null || player == null)
                return;
            TimeStop.Cancel(); // the cut-in owns Time.timeScale while it plays
            var animator = player.GetComponentInChildren<Animator>();
            if (animator == null || !animator.isHuman)
                return;
            var go = new GameObject("NowCutIn");
            current = go.AddComponent<NowCutIn>();
            current.Begin(animator);
        }

        private void Begin(Animator source)
        {
            // Film a stand-in, not the live player: the shot is then identical every time, whatever the player is doing
            // (running, jumping, facing left/right). The stand-in stands far below the level in the controller's
            // default (idle) state replayed from its start, always facing +X.
            var animator = BuildStandIn(source);
            modelRoot = animator.transform;
            eyeL = animator.GetBoneTransform(HumanBodyBones.LeftEye) ?? animator.GetBoneTransform(HumanBodyBones.Head);
            eyeR = animator.GetBoneTransform(HumanBodyBones.RightEye) ?? eyeL;
            FindFace(animator.transform);

            savedTimeScale = Time.timeScale > 0f ? Time.timeScale : 1f;
            Time.timeScale = 0f;
            Sfx.Play(SfxId.CutInNow);

            BuildCamera();
            BuildUi();
            Apply(0f);
        }

        // ---------------------------------------------------------------- stand-in model

        private static readonly Vector3 StandInPosition = new Vector3(0f, -500f, 0f);
        private GameObject standIn;

        private Animator BuildStandIn(Animator source)
        {
            standIn = Instantiate(source.gameObject, StandInPosition, Quaternion.Euler(0f, 90f, 0f));
            standIn.name = "NowCutInStandIn";
            standIn.transform.localScale = source.transform.lossyScale;
            foreach (var behaviour in standIn.GetComponentsInChildren<MonoBehaviour>(true))
                behaviour.enabled = false; // auto-blink, face events, ... : the cut-in drives the face itself
            foreach (var col in standIn.GetComponentsInChildren<Collider>(true))
                col.enabled = false;
            var animator = standIn.GetComponent<Animator>();
            animator.applyRootMotion = false;
            animator.updateMode = AnimatorUpdateMode.UnscaledTime; // keeps animating while time is stopped
            animator.Rebind(); // controller default state (idle) from its start, default parameters: same every time
            return animator;
        }

        // ---------------------------------------------------------------- face

        private void FindFace(Transform root)
        {
            foreach (var smr in root.GetComponentsInChildren<SkinnedMeshRenderer>(true))
            {
                var mesh = smr.sharedMesh;
                if (mesh == null || mesh.blendShapeCount == 0)
                    continue;
                // SD Unity-chan names; other models simply keep their face.
                string[] names = { "face.EYE_ang", "face.BLW_ang", "face.MTH_sml2" };
                float[] weights = { 70f, 100f, 60f };
                var shapes = new System.Collections.Generic.List<int>();
                var w = new System.Collections.Generic.List<float>();
                for (int i = 0; i < names.Length; i++)
                {
                    int index = mesh.GetBlendShapeIndex(names[i]);
                    if (index >= 0) { shapes.Add(index); w.Add(weights[i]); }
                }
                if (shapes.Count == 0)
                    continue;
                face = smr;
                faceShapes = shapes.ToArray();
                faceWeights = w.ToArray();
                blinkShape = mesh.GetBlendShapeIndex("face.EYE_blk");
                return;
            }
        }

        private void ApplyFace()
        {
            if (face == null)
                return;
            var mesh = face.sharedMesh;
            for (int i = 0; i < mesh.blendShapeCount; i++)
            {
                string n = mesh.GetBlendShapeName(i);
                if (n.StartsWith("face.EYE", System.StringComparison.Ordinal) || n.StartsWith("face.BLW", System.StringComparison.Ordinal)
                    || n.StartsWith("face.MTH", System.StringComparison.Ordinal))
                    face.SetBlendShapeWeight(i, 0f);
            }
            for (int i = 0; i < faceShapes.Length; i++)
                face.SetBlendShapeWeight(faceShapes[i], faceWeights[i]);
            if (blinkShape >= 0)
                face.SetBlendShapeWeight(blinkShape, 0f); // eyes wide open
        }

        // ---------------------------------------------------------------- eye camera

        private void BuildCamera()
        {
            rt = new RenderTexture(RtWidth, RtHeight, 24, RenderTextureFormat.ARGB32) { antiAliasing = 4, name = "NowCutInRT" };
            var go = new GameObject("NowCutInCamera");
            go.transform.SetParent(transform, false);
            eyeCam = go.AddComponent<Camera>();
            eyeCam.targetTexture = rt;
            eyeCam.clearFlags = CameraClearFlags.SolidColor;
            eyeCam.backgroundColor = new Color(0f, 0f, 0f, 0f); // only the face is drawn: the band shows through
            eyeCam.nearClipPlane = 0.02f;
            eyeCam.farClipPlane = 0.75f;                          // cuts the world behind the head
            eyeCam.fieldOfView = 11f;
            eyeCam.aspect = (float)RtWidth / RtHeight;
            eyeCam.depth = -10f;
            PlaceCamera();
        }

        private Vector3 EyeMid => (eyeL.position + eyeR.position) * 0.5f;

        private void PlaceCamera()
        {
            if (eyeCam == null || eyeL == null)
                return;
            Vector3 forward = modelRoot.forward;
            forward.y = 0f;
            if (forward.sqrMagnitude < 0.01f)
                forward = Vector3.right;
            forward.Normalize();
            // Slightly toward the side camera so the face reads in three-quarter view; zooms in over the cut-in.
            Vector3 toward = (forward + Vector3.back * 0.35f).normalized;
            float dist = Mathf.Lerp(0.52f, 0.44f, Mathf.Clamp01(t / BandOut));
            Vector3 mid = EyeMid + Vector3.up * 0.012f;
            eyeCam.transform.position = mid + toward * dist;
            eyeCam.transform.rotation = Quaternion.LookRotation(mid - eyeCam.transform.position, Vector3.up);
        }

        // ---------------------------------------------------------------- UI

        private void BuildUi()
        {
            var canvas = gameObject.AddComponent<Canvas>();
            canvas.renderMode = RenderMode.ScreenSpaceOverlay;
            canvas.sortingOrder = 40;
            var scaler = gameObject.AddComponent<CanvasScaler>();
            scaler.uiScaleMode = CanvasScaler.ScaleMode.ScaleWithScreenSize;
            scaler.referenceResolution = new Vector2(1920f, 1080f);
            scaler.matchWidthOrHeight = 0.5f;

            dim = FullScreen("Dim", new Color(0.02f, 0.01f, 0.06f, 0f));

            band = Rect("Band", transform, new Vector2(2700f, BandHeight));
            band.localRotation = Quaternion.Euler(0f, 0f, BandAngle);
            band.anchoredPosition = new Vector2(0f, 90f);
            var bandBack = band.gameObject.AddComponent<Image>();
            bandBack.color = new Color(0.86f, 0.10f, 0.16f);
            bandBack.raycastTarget = false;
            band.gameObject.AddComponent<Mask>().showMaskGraphic = true;

            lines = Rect("SpeedLines", band, new Vector2(2700f, BandHeight)).gameObject.AddComponent<RawImage>();
            lines.texture = SpeedLineTexture();
            lines.color = new Color(1f, 0.85f, 0.3f, 0.55f);
            lines.uvRect = new Rect(0f, 0f, 3f, 1f);
            lines.raycastTarget = false;

            eyeImage = Rect("Eyes", band, new Vector2(BandHeight * RtWidth / RtHeight, BandHeight));
            var eyes = eyeImage.gameObject.AddComponent<RawImage>();
            eyes.texture = rt;
            eyes.raycastTarget = false;

            streak = Rect("Streak", band, new Vector2(1400f, 10f));
            var s = streak.gameObject.AddComponent<Image>();
            s.sprite = GlintSprite();
            s.color = Color.white;
            s.raycastTarget = false;

            glintA = Glint("GlintA", 190f);
            glintB = Glint("GlintB", 90f);

            // Stripe edges (drawn over the mask edge)
            Stripe("EdgeTop", BandHeight * 0.5f, new Color(1f, 0.85f, 0.15f));
            Stripe("EdgeBottom", -BandHeight * 0.5f, new Color(1f, 0.85f, 0.15f));

            text = Rect("Now", transform, new Vector2(1400f, 220f));
            text.anchoredPosition = new Vector2(330f, -170f);
            text.localRotation = Quaternion.Euler(0f, 0f, BandAngle + 2f);
            var label = text.gameObject.AddComponent<Text>();
            label.alignment = TextAnchor.MiddleCenter;
            label.fontSize = 170;
            label.fontStyle = FontStyle.BoldAndItalic;
            label.color = GoldUiTheme.Cream;
            label.resizeTextForBestFit=true;label.resizeTextMinSize=64;label.resizeTextMaxSize=140;
            label.raycastTarget = false;
            label.horizontalOverflow = HorizontalWrapMode.Overflow;
            label.verticalOverflow = VerticalWrapMode.Overflow;
            var outline = text.gameObject.AddComponent<Outline>();
            outline.effectColor = GoldUiTheme.Panel;
            outline.effectDistance = new Vector2(7f, -7f);
            var shadow = text.gameObject.AddComponent<Shadow>();
            shadow.effectColor = new Color(0f, 0f, 0f, 0.8f);
            shadow.effectDistance = new Vector2(14f, -14f);
            var localized = text.gameObject.AddComponent<LocalizedText>();
            localized.SetKey(TextKey);
            if (!GameSystems.Exists)
            {
                LocalizationTable.TryGet("ko", TextKey, out string fallback);
                label.font = FontProvider.Get("ko");
                localized.SetText(fallback ?? "NOW!");
            }

            flash = FullScreen("Flash", new Color(1f, 1f, 1f, 0f));
            bandGraphics = band.GetComponentsInChildren<Graphic>(true);
        }

        private RectTransform Glint(string name, float size)
        {
            var r = Rect(name, band, new Vector2(size, size));
            var img = r.gameObject.AddComponent<Image>();
            img.sprite = GlintSprite();
            img.color = Color.white;
            img.raycastTarget = false;
            r.localScale = Vector3.zero;
            return r;
        }

        private void Stripe(string name, float y, Color color)
        {
            var r = Rect(name, band, new Vector2(2700f, 14f));
            r.anchoredPosition = new Vector2(0f, y);
            var img = r.gameObject.AddComponent<Image>();
            img.color = color;
            img.raycastTarget = false;
        }

        private Image FullScreen(string name, Color color)
        {
            var r = Rect(name, transform, Vector2.zero);
            r.anchorMin = Vector2.zero;
            r.anchorMax = Vector2.one;
            r.sizeDelta = new Vector2(400f, 400f);
            var img = r.gameObject.AddComponent<Image>();
            img.color = color;
            img.raycastTarget = false;
            return img;
        }

        private static RectTransform Rect(string name, Transform parent, Vector2 size)
        {
            var r = new GameObject(name, typeof(RectTransform)).GetComponent<RectTransform>();
            r.SetParent(parent, false);
            r.sizeDelta = size;
            return r;
        }

        // ---------------------------------------------------------------- timeline

        private void LateUpdate()
        {
            t += Time.unscaledDeltaTime;
            ApplyFace();
            PlaceCamera();
            Apply(t);
            if (t >= Total)
                Finish();
        }

        private void Apply(float time)
        {
            // Dim + white flash
            float dimK = time < 0.1f ? time / 0.1f : time > BandOut ? 1f - Mathf.Clamp01((time - BandOut) / (Total - BandOut)) : 1f;
            dim.color = new Color(dim.color.r, dim.color.g, dim.color.b, 0.5f * dimK);
            float flashK = time < 0.12f ? 1f - time / 0.12f : time >= GlintAt && time < GlintAt + 0.08f ? 0.35f * (1f - (time - GlintAt) / 0.08f) : 0f;
            flash.color = new Color(1f, 1f, 1f, 0.75f * flashK);

            // Band: sweeps in from the right, holds (slow drift), leaves to the left
            float x;
            if (time < BandIn) x = Mathf.Lerp(2600f, 0f, EaseOut(time / BandIn));
            else if (time < BandOut) x = Mathf.Lerp(0f, -40f, (time - BandIn) / (BandOut - BandIn));
            else x = Mathf.Lerp(-40f, -2800f, EaseIn(Mathf.Clamp01((time - BandOut) / (Total - BandOut))));
            band.anchoredPosition = new Vector2(x, 90f);
            float open = time < BandIn ? Mathf.Lerp(0.15f, 1f, EaseOut(time / BandIn)) : 1f;
            band.localScale = new Vector3(1f, open, 1f);
            lines.uvRect = new Rect(time * 4f, 0f, 3f, 1f);
            eyeImage.localScale = Vector3.one * Mathf.Lerp(1.12f, 1f, EaseOut(Mathf.Clamp01(time / 0.4f)));

            // Glint on the eye nearest the camera, then the other ("pa-ching")
            PlaceOnEye(glintA, NearEye(true));
            PlaceOnEye(glintB, NearEye(false));
            glintA.localScale = Vector3.one * GlintScale(time - GlintAt);
            glintB.localScale = Vector3.one * 0.8f * GlintScale(time - GlintAt - 0.07f);
            glintA.localRotation = Quaternion.Euler(0f, 0f, -BandAngle + 45f * Mathf.Clamp01((time - GlintAt) / 0.35f));
            glintB.localRotation = Quaternion.Euler(0f, 0f, -BandAngle - 30f * Mathf.Clamp01((time - GlintAt) / 0.35f));
            float sk = time - GlintAt;
            streak.anchoredPosition = glintA.anchoredPosition;
            streak.localScale = sk < 0f ? Vector3.zero : new Vector3(Mathf.Clamp01(sk / 0.06f) * (1f + sk), Mathf.Max(0f, 1f - sk / 0.3f), 1f);

            // "Now!": slams in with an overshoot, shakes a little, leaves with the band
            float tk = time - TextAt;
            float ts = tk < 0f ? 0f : tk < 0.08f ? Mathf.Lerp(2.4f, 0.9f, tk / 0.08f) : tk < 0.16f ? Mathf.Lerp(0.9f, 1.05f, (tk - 0.08f) / 0.08f) : 1f;
            if (time > BandOut) ts *= 1f - Mathf.Clamp01((time - BandOut) / (Total - BandOut));
            text.localScale = Vector3.one * ts;
            float shake = tk > 0f && tk < 0.25f ? (1f - tk / 0.25f) * 10f : 0f;
            text.anchoredPosition = new Vector2(330f + Random.Range(-shake, shake), -170f + Random.Range(-shake, shake));
        }

        private static float GlintScale(float k)
        {
            if (k < 0f) return 0f;
            if (k < 0.07f) return Mathf.Lerp(0f, 1.5f, k / 0.07f);
            if (k < 0.16f) return Mathf.Lerp(1.5f, 0.9f, (k - 0.07f) / 0.09f);
            if (k < 0.45f) return 0.9f + 0.08f * Mathf.Sin(k * 40f);
            return Mathf.Max(0f, 0.9f - (k - 0.45f) * 3f);
        }

        private Transform NearEye(bool near)
        {
            if (eyeCam == null) return eyeL;
            float dl = (eyeL.position - eyeCam.transform.position).sqrMagnitude;
            float dr = (eyeR.position - eyeCam.transform.position).sqrMagnitude;
            return (dl <= dr) == near ? eyeL : eyeR;
        }

        /// <summary>Projects an eye bone through the eye camera onto the RawImage (band space).</summary>
        private void PlaceOnEye(RectTransform r, Transform eye)
        {
            if (eye == null || eyeCam == null)
                return;
            Vector3 v = eyeCam.WorldToViewportPoint(eye.position);
            Vector2 size = eyeImage.sizeDelta * eyeImage.localScale.x;
            r.anchoredPosition = eyeImage.anchoredPosition + new Vector2((v.x - 0.5f) * size.x, (v.y - 0.5f) * size.y + 6f);
        }

        private static float EaseOut(float x) => 1f - (1f - x) * (1f - x);
        private static float EaseIn(float x) => x * x;

        // ---------------------------------------------------------------- end

        private void Finish()
        {
            if (current == this)
                current = null;
            Destroy(gameObject);
        }

        private void OnDestroy()
        {
            if (current == this)
                current = null;
            if (Time.timeScale == 0f)
                Time.timeScale = savedTimeScale; // only this cut-in stops time (pause is refused while it plays)
            if (standIn != null)
                Destroy(standIn);
            if (eyeCam != null)
                eyeCam.targetTexture = null;
            if (rt != null)
                rt.Release();
            Destroy(rt);
        }

        // ---------------------------------------------------------------- procedural sprites

        private static Texture2D speedLines;
        private static Sprite glint;

        /// <summary>Horizontal comic speed lines (tiles in X).</summary>
        private static Texture2D SpeedLineTexture()
        {
            if (speedLines != null)
                return speedLines;
            const int w = 512, h = 128;
            speedLines = new Texture2D(w, h, TextureFormat.RGBA32, false) { wrapMode = TextureWrapMode.Repeat, name = "CutInSpeedLines" };
            var px = new Color32[w * h];
            var rnd = new System.Random(5);
            for (int i = 0; i < 26; i++)
            {
                int y = rnd.Next(2, h - 2);
                int thick = rnd.Next(1, 4);
                int start = rnd.Next(0, w);
                int len = rnd.Next(w / 4, w * 3 / 4);
                for (int k = 0; k < len; k++)
                {
                    int x = (start + k) % w;
                    float taper = Mathf.Sin(Mathf.PI * k / len);
                    for (int dy = -thick; dy <= thick; dy++)
                    {
                        int yy = Mathf.Clamp(y + dy, 0, h - 1);
                        float a = taper * (1f - Mathf.Abs(dy) / (thick + 1f));
                        var c = px[yy * w + x];
                        byte v = (byte)Mathf.Max(c.a, a * 255f);
                        px[yy * w + x] = new Color32(255, 255, 255, v);
                    }
                }
            }
            speedLines.SetPixels32(px);
            speedLines.Apply();
            return speedLines;
        }

        /// <summary>Four-point star with a soft glow (white; tinted by the Image colour).</summary>
        private static Sprite GlintSprite()
        {
            if (glint != null)
                return glint;
            const int n = 128;
            var tex = new Texture2D(n, n, TextureFormat.RGBA32, false) { wrapMode = TextureWrapMode.Clamp, name = "CutInGlint" };
            var px = new Color32[n * n];
            for (int y = 0; y < n; y++)
            for (int x = 0; x < n; x++)
            {
                float u = (x + 0.5f) / n * 2f - 1f, v = (y + 0.5f) / n * 2f - 1f;
                float ax = Mathf.Abs(u), ay = Mathf.Abs(v);
                float star = Mathf.Max(Mathf.Clamp01(1f - ax / Mathf.Max(0.001f, 0.16f * (1f - ay))), Mathf.Clamp01(1f - ay / Mathf.Max(0.001f, 0.16f * (1f - ax))));
                star *= Mathf.Clamp01(1f - Mathf.Max(ax, ay));
                float glow = Mathf.Clamp01(1f - Mathf.Sqrt(u * u + v * v) / 0.45f);
                float a = Mathf.Clamp01(star * 1.6f + glow * glow * 0.8f);
                px[y * n + x] = new Color32(255, 255, 255, (byte)(a * 255f));
            }
            tex.SetPixels32(px);
            tex.Apply();
            glint = Sprite.Create(tex, new Rect(0, 0, n, n), new Vector2(0.5f, 0.5f), 100f);
            return glint;
        }
    }
}
