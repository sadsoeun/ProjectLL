using UnityEngine;
using UnityEngine.UI;

namespace MyGame
{
    /// <summary>Compact icon HUD; keeps the scene's existing text bindings and menu navigation.</summary>
    [DisallowMultipleComponent]
    public sealed class HudPresentation : MonoBehaviour
    {
        private RectTransform stats, chapter, timer, pause;
        private RectTransform lives, coins, time, stage, slots;
        private bool initialized, visible = true;
        public Button PauseButton { get; private set; }
        private static readonly Color Back = new Color(.035f,.038f,.045f,.92f);

        public static HudPresentation Apply(Transform root)
        {
            var hud = root.GetComponent<HudPresentation>();
            if (hud == null) hud = root.gameObject.AddComponent<HudPresentation>();
            hud.Initialize(); hud.Layout();
            return hud;
        }

        public void Initialize()
        {
            if (initialized) return;
            lives = transform.Find("LivesText") as RectTransform;
            coins = transform.Find("CoinsText") as RectTransform;
            time = transform.Find("TimeText") as RectTransform;
            stage = transform.Find("StageNameText") as RectTransform;
            slots = transform.Find("SpecialSlots") as RectTransform;
            if (lives == null || coins == null || time == null || stage == null || slots == null) return;
            initialized = true;
            stats = Plate("HudStats"); chapter = Plate("HudChapter"); timer = Plate("HudTimer");
            Icon(stats, "Portrait", new Vector2(8,4), new Vector2(76,76));
            Icon(stats, "Coin", new Vector2(182,-10), new Vector2(48,48));
            var clock = GoldUiTheme.Rect("Clock",timer);
            GoldUiTheme.Place(clock,new Vector2(0,.5f),new Vector2(0,.5f),new Vector2(14,0),new Vector2(32,32));
            clock.gameObject.AddComponent<HudGlyph>().kind = HudGlyph.Kind.Clock;
            pause = GoldUiTheme.Rect("HudPauseButton",transform);
            var image = pause.gameObject.AddComponent<Image>(); image.color = Color.white;
            PauseButton = pause.gameObject.AddComponent<Button>(); PauseButton.targetGraphic = image;
            GoldUiTheme.StyleButton(PauseButton);
            var colors = PauseButton.colors; colors.normalColor = Back; PauseButton.colors = colors;
            // Esc / gamepad Start remain the keyboard/controller route. Do not steal jump/submit focus.
            PauseButton.navigation = new Navigation { mode = Navigation.Mode.None };
            var bars = GoldUiTheme.Rect("PauseGlyph",pause);
            GoldUiTheme.Place(bars,new Vector2(.5f,.5f),new Vector2(.5f,.5f),Vector2.zero,new Vector2(30,32));
            bars.gameObject.AddComponent<HudGlyph>().kind = HudGlyph.Kind.Pause;
            PauseButton.onClick.AddListener(() => GetComponent<InGameView>()?.RequestPause());
            Style(lives,36,TextAnchor.MiddleLeft); Style(coins,34,TextAnchor.MiddleLeft);
            Style(time,36,TextAnchor.MiddleRight); Style(stage,30,TextAnchor.MiddleCenter);
            SetVisible(visible);
        }

        private RectTransform Plate(string name)
        {
            var rt = GoldUiTheme.Rect(name,transform); rt.SetAsFirstSibling();
            var image = rt.gameObject.AddComponent<Image>(); image.color = Back; image.raycastTarget = false;
            var frame = GoldUiTheme.Frame(rt); frame.diamond = false; frame.color = new Color(.79f,.65f,.42f,.7f);
            return rt;
        }

        private static void Icon(Transform parent,string name,Vector2 position,Vector2 size)
        {
            var rt = GoldUiTheme.Rect(name,parent);
            GoldUiTheme.Place(rt,Vector2.up,Vector2.up,position,size);
            var image = rt.gameObject.AddComponent<Image>(); image.sprite = Resources.Load<Sprite>("HudArt/"+name);
            image.preserveAspect = true; image.raycastTarget = false;
        }

        private static void Style(RectTransform rt,int size,TextAnchor alignment)
        {
            var text = rt.GetComponent<Text>(); text.fontSize = size;
            text.resizeTextForBestFit = true; text.resizeTextMinSize = 22; text.resizeTextMaxSize = size;
            text.fontStyle = FontStyle.Bold; text.color = new Color(1,.95f,.84f);
            text.alignment = alignment; text.raycastTarget = false;
        }

        private void OnRectTransformDimensionsChange() { if (initialized) Layout(); }

        public void Layout()
        {
            if (!initialized) return;
            float width = ((RectTransform)transform).rect.width;
            const float margin = 32, top = -28, height = 72;
            bool narrow = width < 1240;
            float middleWidth = narrow ? Mathf.Min(520,width-64) : Mathf.Min(520,width-780);
            float middleY = narrow ? -116 : top;
            Place(stats,0,margin,top,340,height);
            Place(chapter,.5f,0,middleY,middleWidth,height);
            Place(timer,1,-margin-86,top,220,height);
            Place(pause,1,-margin,top,72,height);
            Place(lives,0,margin+90,top,86,height);
            Place(coins,0,margin+238,top,88,height);
            Place(time,1,-margin-102,top,155,height);
            Place(stage,.5f,0,middleY-2,middleWidth-24,35);
            Place(slots,.5f,0,middleY-44,104,22);
            for (int i=0;i<slots.childCount;i++)
            {
                var rt=slots.GetChild(i) as RectTransform;
                if(rt!=null) GoldUiTheme.Place(rt,new Vector2(0,.5f),new Vector2(.5f,.5f),new Vector2(16+i*36,0),new Vector2(22,22));
            }
        }

        private static void Place(RectTransform rt,float anchor,float x,float y,float w,float h)
        { GoldUiTheme.Place(rt,new Vector2(anchor,1),new Vector2(anchor,1),new Vector2(x,y),new Vector2(w,h)); }

        public void SetVisible(bool value)
        {
            visible=value;
            if(!initialized)return;
            stats.gameObject.SetActive(value); chapter.gameObject.SetActive(value);
            timer.gameObject.SetActive(value); pause.gameObject.SetActive(value);
        }

        public void SetPauseAvailable(bool value)
        { if (PauseButton != null) PauseButton.interactable = value; }
    }
}
