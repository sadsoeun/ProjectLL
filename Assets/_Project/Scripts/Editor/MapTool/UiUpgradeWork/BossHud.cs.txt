using UnityEngine;
using UnityEngine.UI;

namespace MyGame
{
    /// <summary>
    /// Boss HP bar: localized boss name above one segment per HP. Built in code on its own overlay canvas (no scene or
    /// prefab changes needed), created by BossController when the fight starts and destroyed when it ends.
    /// </summary>
    public sealed class BossHud : MonoBehaviour
    {
        private static readonly Color Full = new Color(0.95f, 0.25f, 0.4f);
        private static readonly Color Empty = new Color(0.15f, 0.12f, 0.16f, 0.9f);

        private Image[] segments;
        private RectTransform root;
        private float pulse;
        private InGameView gameView;

        public static BossHud Create(string nameKey, int maxHp)
        {
            var go = new GameObject("BossHud");
            var canvas = go.AddComponent<Canvas>();
            canvas.renderMode = RenderMode.ScreenSpaceOverlay;
            canvas.sortingOrder = 30;
            var scaler = go.AddComponent<CanvasScaler>();
            scaler.uiScaleMode = CanvasScaler.ScaleMode.ScaleWithScreenSize;
            scaler.referenceResolution = new Vector2(1920f, 1080f);
            scaler.matchWidthOrHeight = 0.5f;
            var hud = go.AddComponent<BossHud>();
            hud.Build(nameKey, Mathf.Max(1, maxHp));
            return hud;
        }

        private void Build(string nameKey, int maxHp)
        {
            root = new GameObject("Panel", typeof(RectTransform)).GetComponent<RectTransform>();
            root.SetParent(transform, false);
            root.anchorMin = root.anchorMax = new Vector2(0.5f, 1f);
            root.pivot = new Vector2(0.5f, 1f);
            root.anchoredPosition = new Vector2(0f, -120f);
            root.sizeDelta = new Vector2(760f, 100f);
            var plate=root.gameObject.AddComponent<Image>();plate.color=GoldUiTheme.Panel;plate.raycastTarget=false;
            GoldUiTheme.Frame(root).diamond=false;
            gameView=Object.FindFirstObjectByType<InGameView>();

            var nameGo = new GameObject("Name", typeof(RectTransform), typeof(Text));
            var nameRect = nameGo.GetComponent<RectTransform>();
            nameRect.SetParent(root, false);
            nameRect.anchorMin = new Vector2(0f, 1f);
            nameRect.anchorMax = new Vector2(1f, 1f);
            nameRect.pivot = new Vector2(0.5f, 1f);
            nameRect.sizeDelta = new Vector2(0f, 48f);
            var text = nameGo.GetComponent<Text>();
            text.alignment = TextAnchor.MiddleCenter;
            UiKit.TextStyle(nameRect,UiKit.HudSize,TextAnchor.MiddleCenter);
            text.color = GoldUiTheme.Cream;
            text.raycastTarget = false;
            var localized = nameGo.AddComponent<LocalizedText>();
            localized.SetKey(nameKey);
            if (!GameSystems.Exists)
            {
                LocalizationTable.TryGet(LocalizationTable.DefaultLanguage, nameKey, out string fallback);
                localized.SetText(fallback ?? nameKey);
            }

            var back = new GameObject("Back", typeof(RectTransform), typeof(Image));
            var backRect = back.GetComponent<RectTransform>();
            backRect.SetParent(root, false);
            backRect.anchorMin = new Vector2(0f, 0f);
            backRect.anchorMax = new Vector2(1f, 0f);
            backRect.pivot = new Vector2(0.5f, 0f);
            backRect.sizeDelta = new Vector2(0f, 34f);
            var backImage = back.GetComponent<Image>();
            backImage.color = new Color(0f, 0f, 0f, 0.65f);
            backImage.raycastTarget = false;

            segments = new Image[maxHp];
            const float pad = 4f;
            float width = (760f - pad * (maxHp + 1)) / maxHp;
            for (int i = 0; i < maxHp; i++)
            {
                var seg = new GameObject("Hp" + i, typeof(RectTransform), typeof(Image));
                var rect = seg.GetComponent<RectTransform>();
                rect.SetParent(backRect, false);
                rect.anchorMin = rect.anchorMax = new Vector2(0f, 0.5f);
                rect.pivot = new Vector2(0f, 0.5f);
                rect.anchoredPosition = new Vector2(pad + i * (width + pad), 0f);
                rect.sizeDelta = new Vector2(width, 24f);
                segments[i] = seg.GetComponent<Image>();
                segments[i].raycastTarget = false;
                segments[i].color = Full;
            }
        }

        /// <summary>Shows <paramref name="hp"/> filled segments. A drop pulses the bar.</summary>
        public void SetHp(int hp)
        {
            if (segments == null)
                return;
            for (int i = 0; i < segments.Length; i++)
                segments[i].color = i < hp ? Full : Empty;
            pulse = 1f;
        }

        private void Update()
        {
            if (root == null)return;
            if(gameView==null)gameView=Object.FindFirstObjectByType<InGameView>();
            bool visible=(gameView==null||gameView.IsHudVisible)&&(!GameSystems.Exists||!GameSystems.Instance.Popups.IsBlocking);
            root.gameObject.SetActive(visible);
            float width=((RectTransform)transform).rect.width;
            root.anchoredPosition=new Vector2(0,width<1240?-220:-120);
            pulse = Mathf.MoveTowards(pulse, 0f, Time.unscaledDeltaTime * 3f);
            root.localScale = Vector3.one * Mathf.Min(1,Mathf.Max(.1f,((RectTransform)transform).rect.width/900f)) * (1f + 0.04f * pulse);
        }
    }
}
